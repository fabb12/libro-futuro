% Immagine con segnaposto: se il file non esiste ancora in images/ compare un
% riquadro con la descrizione, e il libro compila lo stesso. Quando il file
% viene aggiunto, l'immagine appare da sola.
% Uso: \immagine{images/nome.png}{descrizione di cosa deve mostrare}
% Dimensionamento delle immagini: tutta la larghezza del testo, ma mai piu'
% alte di 0.87 dell'altezza utile, cosi' immagine e didascalia stanno sempre
% nella stessa pagina. Un'immagine alta finisce da sola in una pagina che
% riempie quasi per intero; una bassa si incastra nel testo senza vuoti.
\setkeys{Gin}{keepaspectratio,height=0.87\textheight}
\newcommand{\immagine}[2]{%
	\IfFileExists{#1}%
		{\includegraphics[width=\textwidth]{#1}}%
		{\fbox{\parbox{0.8\textwidth}{\centering\small\vspace{1.5em}%
			[IMMAGINE DA INSERIRE: #2]\\[0.4em]\texttt{\detokenize{#1}}\vspace{1.5em}}}}%
}

% =============================================================================
% IMMAGINI DA SOLE IN UNA PAGINA
% Una figura che finisce da sola in una pagina viene ingrandita il più
% possibile senza deformarla: può uscire di 12 mm per lato nei margini e
% occupa tutta l'altezza utile, tolto lo spazio della didascalia (misurata).
% Uso: \figurasola{images/file.png}{didascalia}{etichetta}
% Se il file non esiste ancora compare un segnaposto e il libro compila lo stesso.
% =============================================================================
\newlength{\figsolacapht}
\newlength{\figsolaextra}
\newsavebox{\figsolabox}
\setlength{\figsolaextra}{24mm}
\newcommand{\figurasola}[3]{%
	\begin{figure}[p]
		\centering
		\sbox{\figsolabox}{\parbox{\dimexpr\textwidth-1cm\relax}{\scriptsize\sffamily\centering\textbf{Figura 00.00} #2}}\setlength{\figsolacapht}{\dimexpr\ht\figsolabox+\dp\figsolabox\relax}%
		\IfFileExists{#1}%
			{\makebox[\textwidth][c]{\includegraphics[width=\dimexpr\textwidth+\figsolaextra\relax,height=\dimexpr\textheight-\figsolacapht-\abovecaptionskip-\belowcaptionskip-14pt\relax,keepaspectratio]{#1}}}%
			{\fbox{\parbox{0.8\textwidth}{\centering\small\vspace{3em}%
				[IMMAGINE DA INSERIRE]\\[0.4em]\texttt{\detokenize{#1}}\vspace{3em}}}}%
		\caption{#2}%
		\label{#3}%
	\end{figure}%
}
